## Script corregido (LC_ALL=C fuerza la salida en ingles solo para ese comando)
cloud@ubuntu-ollama:~$ diff reporte_sistema_original.sh reporte_sistema.sh
8c8
< lscpu | grep "Model name"
---
> LC_ALL=C lscpu | grep "Model name"   # LC_ALL=C: con Ubuntu en espanol lscpu imprime "Nombre del modelo"

cloud@ubuntu-ollama:~$ cat reporte_sistema.sh
#!/bin/bash
echo "=== REPORTE DEL SISTEMA ==="
date
hostname
uname -a
echo "--- CPU ---"
nproc
LC_ALL=C lscpu | grep "Model name"   # LC_ALL=C: con Ubuntu en espanol lscpu imprime "Nombre del modelo"
echo "--- RAM ---"
free -h
echo "--- DISCO ---"
df -h /
echo "--- OLLAMA ---"
systemctl is-active ollama

cloud@ubuntu-ollama:~$ chmod +x reporte_sistema.sh

cloud@ubuntu-ollama:~$ ls -l reporte_sistema.sh
-rwxr-xr-x 1 cloud cloud 305 sep 24 22:03 reporte_sistema.sh
